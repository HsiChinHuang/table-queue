**Shared section extractor (every section-scoped check below uses it).** A plain `awk '/^## Labels/{...}'` is unsafe in this file: while the template block is unclosed, and even after it closes, the embedded template copy contains its own `## Definition of Done` at `CONTRIBUTING.md:114`, so a naive `awk` range silently reads the *template example* instead of the document section. Create this helper once and call it from the checks:

```bash
cat > /tmp/p009-sect.sh <<'SH'
node -e '
const fs = require("fs");
const want = process.argv[1], lvl = Number(process.argv[2] || 2);
const lines = fs.readFileSync("CONTRIBUTING.md", "utf8").replace(/\r/g, "").split("\n");
let inFence = false, start = -1;
const hashes = "#".repeat(lvl);
for (let i = 0; i < lines.length; i++) {
  const l = lines[i];
  if (/^\s*```/.test(l)) { inFence = !inFence; continue; }
  if (inFence) continue;
  if (start < 0) { if (new RegExp("^" + hashes + " " + want + "\\s*$").test(l)) start = i; }
  else if (/^#{1,6} /.test(l)) { console.log(lines.slice(start + 1, i).join("\n")); process.exit(0); }
}
if (start >= 0) { console.log(lines.slice(start + 1).join("\n")); process.exit(0); }
console.error("FAIL section not found outside fences: " + want); process.exit(1);
' "$1" "${2:-2}"
SH
```
