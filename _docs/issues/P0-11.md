- [ ] **AC-5**: The dependency graph is acyclic (topological sort succeeds).
```bash
python3 << 'EOF'
import re, sys
plan = open("_docs/plan.md").read()
edges = []
in_section = False
for line in plan.split("\n"):
    if "## 1. Dependency Graph" in line:
        in_section = True
        continue
    if in_section and line.startswith("## "):
        break
    if in_section and "├──" in line or "└──" in line:
        parts = line.strip().replace("├──", "").replace("└──", "").strip().split()
        if parts:
            dep = parts[0]
            edges.append(dep)
graph = {}
for e in edges:
    if ":" in e:
        parent, child = e.split(":")
        graph.setdefault(parent.strip(), []).append(child.strip())
def has_cycle(node, visited, rec_stack):
    visited.add(node)
    rec_stack.add(node)
    for neighbor in graph.get(node, []):
        if neighbor not in visited:
            if has_cycle(neighbor, visited, rec_stack):
                return True
        elif neighbor in rec_stack:
            return True
    rec_stack.remove(node)
    return False
visited = set()
cycle = False
for node in graph:
    if node not in visited:
        if has_cycle(node, visited, set()):
            cycle = True
            break
if not cycle:
    print("PASS AC-5 dependency graph is acyclic")
else:
    print("FAIL AC-5 cycle detected in dependency graph")
EOF
```
Measured today: PASS AC-5 dependency graph is acyclic
