    ct2=$(curl -s -o /dev/null -w '%{http_code}' -X POST http://localhost:8011/api/v1/waitlist/$qn2/cancel \

    pkill $be 2>/dev/null ; \
    echo "WITH_RC=0" > /tmp/tq-ac3.log ; \
    [ "$st" != "200" ] && echo "WITH_RC=1" > /tmp/tq-ac3.log ; \
    [ "$ct" != "200" ] && echo "WITH_RC=1" > /tmp/tq-ac3.log ; \

    [ "$stn" = "200" ] && echo "WITHOUT_RC=1" >> /tmp/tq-ac3.log ; \
